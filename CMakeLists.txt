cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(SchoolMonitorNative VERSION 0.1.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8 /W3)
else()
  add_compile_options(-Wall -Wextra -Wno-unused-parameter)
endif()
add_library(sqlite STATIC third_party/sqlite3.c)
target_include_directories(sqlite PUBLIC third_party)
target_compile_definitions(sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0)
if(UNIX)
  target_link_libraries(sqlite PUBLIC ${CMAKE_DL_LIBS} pthread)
  target_compile_options(sqlite PRIVATE -w)
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/Inter.ttf" FONT_HEX HEX)
string(REGEX REPLACE "(..)" "0x\\1," FONT_BYTES "${FONT_HEX}")
configure_file(src/font.cpp.in generated/font.cpp @ONLY)
add_library(monitor_core STATIC src/model.cpp src/pdf.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/font.cpp")
target_include_directories(monitor_core PUBLIC src third_party)
target_link_libraries(monitor_core PUBLIC sqlite)
add_executable(monitor_tests tests/tests.cpp)
target_link_libraries(monitor_tests PRIVATE monitor_core)
enable_testing()
add_test(NAME core COMMAND monitor_tests)
option(MONITOR_BUILD_GUI "Build wxWidgets desktop application" ON)
if(MONITOR_BUILD_GUI)
  include(FetchContent)
  set(wxBUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(wxBUILD_USE_STATIC_RUNTIME ON CACHE BOOL "" FORCE)
  set(wxBUILD_SAMPLES OFF CACHE BOOL "" FORCE)
  set(wxBUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(wxUSE_ZLIB builtin CACHE STRING "" FORCE)
  set(wxUSE_EXPAT builtin CACHE STRING "" FORCE)
  set(wxUSE_LIBWEBP OFF CACHE STRING "" FORCE)
  set(wxUSE_WEBVIEW OFF CACHE BOOL "" FORCE)
  set(wxUSE_MEDIACTRL OFF CACHE BOOL "" FORCE)
  set(wxUSE_STC OFF CACHE BOOL "" FORCE)
  set(wxUSE_AUI OFF CACHE BOOL "" FORCE)
  set(wxUSE_LIBTIFF OFF CACHE STRING "" FORCE)
  set(wxUSE_LIBJPEG OFF CACHE STRING "" FORCE)
  FetchContent_Declare(wx URL https://github.com/wxWidgets/wxWidgets/releases/download/v3.3.4/wxWidgets-3.3.4.tar.bz2 URL_HASH SHA256=815c16cafa4fcbfcf25c5c0a3418309c62257b8a3b6f91d85a1367a134f98c98 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(wx)
  add_executable(SchoolMonitor WIN32 MACOSX_BUNDLE src/app.cpp)
  target_link_libraries(SchoolMonitor PRIVATE monitor_core wx::core wx::base wx::adv)
  set_target_properties(SchoolMonitor PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Школьный мониторинг" MACOSX_BUNDLE_GUI_IDENTIFIER "dev.pepelaz.schoolmonitor.native" MACOSX_BUNDLE_SHORT_VERSION_STRING "0.1.1")
  if(APPLE)
    set_source_files_properties(assets/Inter.ttf PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/Fonts")
    target_sources(SchoolMonitor PRIVATE assets/Inter.ttf)
    set_target_properties(SchoolMonitor PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/src/Info.plist.in")
    target_link_options(SchoolMonitor PRIVATE -Wl,-dead_strip)
  elseif(MSVC)
    target_link_options(SchoolMonitor PRIVATE /OPT:REF /OPT:ICF)
  endif()
endif()
